-- database/014_separate_flexible_survey_responses.sql
CREATE TABLE IF NOT EXISTS ipak_survey_responses (
    kode BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    response_code VARCHAR(40) NOT NULL,
    form_id INT UNSIGNED NOT NULL,
    form_code VARCHAR(80) NOT NULL,
    submitted_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    respondent_name VARCHAR(200) DEFAULT NULL,
    respondent_email VARCHAR(200) DEFAULT NULL,
    respondent_phone VARCHAR(50) DEFAULT NULL,
    metadata_json LONGTEXT DEFAULT NULL,
    PRIMARY KEY (kode),
    UNIQUE KEY uq_ipak_flex_response_code (response_code),
    KEY idx_ipak_flex_form_submitted (form_id, submitted_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
